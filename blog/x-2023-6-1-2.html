[PlayerControl.cs]

using System.Collections;
using System.Collections.Generic;
using UnityEngine;
 
public class PlayerControl : MonoBehaviour
{
 
    public float speed;
     private Rigidbody2D rrigidbody;
    private Animator animator;
    private float inputX, inputY;
    private float stopX, stopY;//记录角色的朝向
    public bool canmoving = true;
    //public GameObject wakvoice;
    void Start()
    {
        rrigidbody = GetComponent<Rigidbody2D>();
        animator = GetComponent<Animator>();
    }
 
    void Update()
    {
        inputX = Input.GetAxisRaw("Horizontal");
        inputY = Input.GetAxisRaw("Vertical");
        //inputX=joystick.Horizontal;
        //inputY=joystick.Vertical;
        Vector2 qwe = new Vector2(inputX, inputY).normalized;
        animator.SetFloat("inputX", inputX);
        animator.SetFloat("inputY", inputY);
        //if(inputX>0||inputY>0)
        //{
        //   wakvoice.SetActive(true);
        //}
        //else
        //{
        //    wakvoice.SetActive(false);
        //}
        rrigidbody.velocity = qwe * speed;
        
 
        if (qwe != Vector2.zero && canmoving)
        {
            animator.SetBool("isMoving", true);
            //stopX = inputX;
            //stopY = inputY;
        }
        else
        {
            animator.SetBool("isMoving", false);
        }
        animator.SetFloat("inputX", inputX);
        animator.SetFloat("inputY", inputY);
 
    }
}

[ChoseButton.cs]

using System.Collections;
using System.Collections.Generic;
using UnityEngine;
 
public class ChoseButton : MonoBehaviour
{
    public GameObject Button;
    public GameObject talkUI;
    public GameObject Voice;
    
    private void OnTriggerEnter2D(Collider2D other)
    {
        if (other.CompareTag("PlayerChose"))
            { 
             Button.SetActive(true);
            }
    }
 
    private void OnTriggerExit2D(Collider2D other)
    {
        if (other.CompareTag("PlayerChose"))
            {
            Button.SetActive(false);
            }
    }
 
    private void Update()
    {
        if (Button.activeSelf && Input.GetKeyDown(KeyCode.Z))
        {   
            Voice.SetActive(true);
            talkUI.SetActive(true);
        }
        
        if (talkUI.activeInHierarchy)//activeSelf
        {
            GameObject.Find("Player").GetComponent<PlayerControl>().enabled =false;
 
        }
        else
        {
            GameObject.Find("Player").GetComponent<PlayerControl>().enabled =true;
            Voice.SetActive(false);
        }
    }
 
}

using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;
public class Dialog : MonoBehaviour
{
 [Header("UI组件")]
public Text textLabel;
//public Image faceImage;
    [Header("声音文件")]
public GameObject Type;
 [Header("文本文件")]
public TextAsset textFile;
public int index;
public float textspeed;
 
    //AudioSource Enter0;
 
bool textFinished;
List<string> textList = new List<string>();
    void Awake()
    {
        GetTextFormFile(textFile);
    }
    private void OnEnable () 
    {
        //textLabel.text = textList[index];
        //index++;
        textFinished = true;
        StartCoroutine(SetTextUI());
 
    }
    // Update is called once per frame
    void Update()
    {
 
        Type.SetActive(true);
 
        if (Input.GetKeyDown(KeyCode.Z) && index == textList.Count)
        {
            Type.SetActive(false);
            gameObject.SetActive(false);
            index = 0;
            
            //Enter0 = GetComponent<AudioSource>();
            //Enter0.Play();
        }
 
        if (textFinished)
        {
            Type.SetActive(false);
 
        }
 
        if (Input.GetKeyDown(KeyCode.Z) && textFinished && gameObject.activeSelf)
        {
            //textLabel.text = textList[index];
            //index++;
 
            StartCoroutine(SetTextUI());
//
 
        }
    }
 
    void GetTextFormFile(TextAsset file) 
    {
        {
            
            textList.Clear();
            index = 0;
 
            var lineDate = file.text.Split('\n');
 
            foreach(var line in lineDate) 
            {
                textList.Add(line);
                Type.SetActive(true);
            }
        }
    }
 
    IEnumerator SetTextUI()
    {   textFinished = false;
        textLabel.text = " ";
        
 
        for (int i = 0; i < textList[index].Length; i++) 
        {
            textLabel.text += textList[index][i];
            Type.SetActive(false);
            yield return new WaitForSeconds(textspeed);    
        }
        textFinished = true;
        index++;
    }
 
}
